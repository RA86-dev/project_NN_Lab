# Extension Building
Here is a example Extension. A extension in this case, is a custom `.js` file that allows you to add more features to the script. (NOTE: This was generated by AI - the Sample Extension).
```javascript
/**
 * Example Extension: Custom Leaky ReLU Activation Layer
 * Allows a small, non-zero gradient (alpha) when the unit is not active.
 */
export default {
  id: "extension_leaky_relu",
  name: "Leaky ReLU Layer Extension",
  author: "Community",
  version: "1.0.0",

  // 1. Register the custom visual block in Blockly
  registerBlocks(Blockly) {
    Blockly.Blocks["extension_leaky_relu_layer"] = {
      init: function () {
        this.appendDummyInput()
          .appendField("Leaky ReLU Alpha")
          .appendField(new Blockly.FieldNumber(0.2, 0, 1, 0.01), "ALPHA");
        
        this.setPreviousStatement(true, "LAYER");
        this.setNextStatement(true, "LAYER");
        this.setColour("#e06c75"); // Custom extension color accent
        this.setTooltip("Leaky ReLU activation layer with configurable alpha coefficient.");
      },
    };
  },

  // 2. Define where it appears in the Toolbox
  // Toolbox items always appear in the new category called 'EXTENSIONS'!
  
  toolboxItems: [
    {
      kind: "block",
      type: "extension_leaky_relu_layer",
    },
  ],

  // 3. Convert the Blockly block into your custom AST structure
  compileBlock(block) {
    if (block.type === "extension_leaky_relu_layer") {
      return {
        type: "extension_leaky_relu_layer",
        alpha: Number(block.getFieldValue("ALPHA") ?? 0.2),
      };
    }
    return null; // Fall through if not this block
  },

  // 4. Convert the AST node into TensorFlow.js code string
  // Note: `${firstInputShape()}` automatically handles appending `, inputShape: [...]` 
  // if this extension block happens to be the very first layer inside a sequential model!
  compileCode(node, firstInputShape) {
    if (node.type === "extension_leaky_relu_layer") {
      const alpha = node.alpha ?? 0.2;
      // Note this has to return a VALID tfJS line.
      return `tf.layers.leakyReLu({ alpha: ${alpha}${firstInputShape()} })`;
    }
    return null; // Fall through if not this block
  },
};
```
